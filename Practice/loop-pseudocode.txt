START

SET total to 0
SET maximum to 0
SET minimum to 0
SET count to 0

DISPLAY "Enter five floating-point values:"

WHILE count is less than 5
    DISPLAY "Value " followed by count + 1
    READ input

    IF input is not a valid floating-point value THEN
        DISPLAY "Please enter a valid floating-point value."
        DISCARD invalid input
    ELSE
        SET value to input

        IF count equals 0 THEN
            SET maximum to value
            SET minimum to value
        ELSE
            IF value is greater than maximum THEN
                SET maximum to value
            END IF

            IF value is less than minimum THEN
                SET minimum to value
            END IF
        END IF

        SET total to total + value
        SET count to count + 1
    END IF
END WHILE

SET average to total divided by 5
SET interest to total multiplied by 0.20

DISPLAY total
DISPLAY average
DISPLAY maximum
DISPLAY minimum
DISPLAY interest

END
