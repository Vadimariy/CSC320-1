START

DECLARE weeklyIncome as double
DECLARE taxRate as double
DECLARE taxWithholding as double

PROMPT user to enter weekly income
READ weeklyIncome

IF weeklyIncome < 500
    taxRate = 0.10
ELSE IF weeklyIncome >= 500 AND weeklyIncome < 1500
    taxRate = 0.15
ELSE IF weeklyIncome >= 1500 AND weeklyIncome < 2500
    taxRate = 0.20
ELSE
    taxRate = 0.30
END IF

taxWithholding = weeklyIncome * taxRate

DISPLAY weekly income
DISPLAY tax withholding
DISPLAY take-home pay (weeklyIncome - taxWithholding)

END
